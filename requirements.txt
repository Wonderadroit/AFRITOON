#!/usr/bin/env python3
import argparse
import os
import subprocess
import tempfile
import yaml
from engine.scene import Scene
from engine.renderer import Renderer


def main():
    p = argparse.ArgumentParser(description="Render an AFRITOON YAML scene to MP4")
    p.add_argument("scene", help="YAML scene file")
    p.add_argument("-o", "--output", default=None)
    args = p.parse_args()
    with open(args.scene, "r", encoding="utf-8") as f:
        scene = Scene.from_dict(yaml.safe_load(f))
    output = args.output or yaml.safe_load(open(args.scene, encoding="utf-8")).get("output", "output/afritoon.mp4")
    os.makedirs(os.path.dirname(output) or ".", exist_ok=True)
    renderer = Renderer(scene)
    with tempfile.TemporaryDirectory(prefix="afritoon_") as tmp:
        for i in range(int(scene.duration * scene.fps)):
            frame = renderer.frame(i / scene.fps)
            frame.save(os.path.join(tmp, f"frame_{i:06d}.png"))
        cmd = ["ffmpeg", "-y", "-framerate", str(scene.fps), "-i", os.path.join(tmp, "frame_%06d.png"), "-c:v", "libx264", "-pix_fmt", "yuv420p", "-movflags", "+faststart", output]
        subprocess.run(cmd, check=True)
    print(f"Rendered: {output}")


if __name__ == "__main__":
    main()
